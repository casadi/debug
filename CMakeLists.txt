cmake_minimum_required(VERSION 3.15)
project(shell_compiler_args C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(WITH_THREAD "Define CASADI_WITH_THREAD" ON)
option(CANDIDATE "Test candidate/ instead of src/" OFF)
if(CANDIDATE)
  add_compile_definitions(HARNESS_CANDIDATE)
endif()

if(MSVC)
  add_compile_options(/W4 /wd4996)
else()
  add_compile_options(-Wall -Wextra)
endif()

add_executable(argv_dump tests/argv_dump.c)
add_executable(harness tests/harness.cpp)
target_include_directories(harness PRIVATE stubs src)
if(WITH_THREAD)
  find_package(Threads REQUIRED)
  target_compile_definitions(harness PRIVATE CASADI_WITH_THREAD)
  target_link_libraries(harness Threads::Threads)
endif()
if(WIN32)
  target_link_libraries(argv_dump shell32)
  target_link_libraries(harness shell32)
endif()
if(UNIX)
  target_link_libraries(harness ${CMAKE_DL_LIBS})
endif()
